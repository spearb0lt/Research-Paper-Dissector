# Base dependencies. Everything here must fit inside a Vercel Python function,
# whose uncompressed bundle limit is 500 MB. Anything heavier belongs in
# requirements-server.txt, which only the Docker, Render and local targets
# install. Keeping that line means a paper parsed on a laptop can still be
# served, searched and browsed from a serverless deployment.

# Web layer
fastapi>=0.115.6,<1.0
uvicorn[standard]>=0.34.0,<1.0
pydantic>=2.10.4,<3.0
python-multipart>=0.0.20
python-dotenv>=1.0.1

# PDF reading. PyMuPDF does structure, text, figures and page rendering;
# pdfplumber does ruled tables, which PyMuPDF has no equivalent for. Together
# they are about 25 MB and need no model weights at all.
pymupdf>=1.24.0
pdfplumber>=0.11.0

# Vectors. Required, not optional: retrieval scoring, BM25 and the dense
# similarity search all go through it.
numpy>=2.0.0

# Local embeddings with no PyTorch. onnxruntime plus a quantised MiniLM is
# about 160 MB unpacked, against roughly 1 GB for the torch stack, which is
# what makes key free retrieval possible on a serverless deployment.
onnxruntime>=1.20.0
tokenizers>=0.21.0

# AI providers. Groq, OpenAI, OpenRouter, Together, DeepSeek, Mistral,
# Cerebras, SambaNova, xAI, Fireworks, Perplexity, Ollama, LM Studio, llama.cpp
# and vLLM all speak the OpenAI protocol, so one client library covers them all.
openai>=1.59.6
google-genai>=1.2.0
anthropic>=0.45.0

# Optional Postgres back end, used only when DATABASE_URL is set. Without it
# the app falls back to SQLite on the local filesystem.
psycopg[binary]>=3.2.3

# HTTP, for fetching a paper by DOI or arXiv id rather than uploading it.
requests>=2.32.3
